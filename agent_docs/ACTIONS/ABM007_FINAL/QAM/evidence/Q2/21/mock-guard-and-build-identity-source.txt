src/utils/get-user-role.ts
1: import { createClient } from './supabase/server';
2: import { AppRole } from './app-role';
3: 
4: /**
5:  * Fetches the user's role from the user_roles table in the database.
6:  * This is the canonical source of truth for authorization.
7:  * 
8:  * @param userId - The authenticated user's ID from auth.users
9:  * @returns The user's role or null if not found
10:  */
11: export const getUserRole = async (userId: string): Promise<AppRole | null> => {
12:   if (!userId) return null;
13: 
14:   const supabase = await createClient();
15:   
16:   const { data, error } = await supabase
17:     .from('user_roles')
18:     .select('role')
19:     .eq('user_id', userId)
20:     .single();
21: 
22:   if (error || !data) {
23:     console.error('Error fetching user role:', error);
24:     return null;
25:   }
26: 
27:   return data.role as AppRole;
28: };
